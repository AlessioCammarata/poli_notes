#### Exercise 1
Write a bash script that computes the values of a function f(x) for all the triples of integer values stored in a file
- f(x) = 3 ⋅ x2 + 4 ⋅ y + 5 ⋅ z
- The name of the file must be passed from command line 
- Write two versions of the script using while and for statements, respectively
```bash
#!/bin/bash

# Control if the number of parameters is correct (i.e., at least one parameter is needed) 
if [ $# -eq 0 ]
then
   echo "Usage: $0 <file_name>"
   exit 1
fi

#Solution 1
# Using a for loop
flag=1 
# Reads from file one value at a time because the output of the command goes in a list of strings
for val in $(cat $1) 
do 
	if [ $flag -eq 1 ] 
	then 
		let f=3*val*val
	elif [ $flag -eq 2 ] 
	then 
		let f=f+4*val 
	elif [ $flag -eq 3 ] 
	then 
		let f=f+5*val 
		flag=0 
		echo -n "$f " 
	fi 
	let flag=flag+1 
done 

#Solution 2
# Using the while loop (reads a line at a time)
while read line # Reads a line and the string is assigned to variable line
do 
	flag=1 
	for val in $line # Parsing the line
	do
		if [ $flag -eq 1 ] 
		then 
			let f=3*val*val 
		elif [ $flag -eq 2 ] 
		then 
			let f=f+4*val 
		elif[ $flag -eq 3 ] 
		then 
			let f=f+5*val 
		fi 
		let flag=flag+1 
		done 
	echo -n "$f " 
done < $1 # Loop on file lines

#Solution 3
#Using the while loop (reads three values at a time)
while read x y z # Values read from file three at a time!
do 
	let f=3*x*x+4*y+5*z 
	echo -n "$f " 
done < $1 # Loop on file lines

exit 0
```

Write a script that receives the path of a source file and a destination path, and
- Check the correct passage of the parameters to the script 
- Make in background a copy of the source file in the destination path 
- Analyze the space occupied on the destination path at regular intervals of one second, displaying on the screen the percentage of progress of the copy operatio
```bash
#!/bin/bash

# Check the number of parameters
if [ $# -ne 2]; then
	echo "Usage $0 <source> <destination>"
	exit 1
fi

# Check the validity of parameters
if [ ! -f $1]; then
	echo "Source is not a valid file"
	exit 1
fi

if [ ! -d $2]; then
	echo "Destination is not a valid directory"
	exit 1
fi

source=$1
destination=$2

size=$(ls -l $source | cut -d " " -f 5)
let "size = size/1024" # Calculate file size in 1KB blocks

# Calculate destination file system size
startUsed = $(df $destination | \ 
				tailn -n 1 | \
				tr -s " " | \
				cut -d " " -f 3)

cp $source $destination & # Copy in background

transferred=0
percentage=0
# Check the state of the copy in background
while [ $transferred -lt $size]; do 
	startUsed = $(df $destination | \ 
				tailn -n 1 | \
				tr -s " " | \
				cut -d " " -f 3)
	let "transferred=currentUsed-startUsed"
	let "percentage=transferred*100/size"
	echo "Progress: $percentage%"
	sleep 1
done

exit 0
```
#### Exercise from exam 2
- Takes 4 arguments (dir1, dir2 e dir3, directory names, and n, an integer number) 
- Finds in dir1 and dir2 all files that have the same name, extension txt and more than n lines 
- Creates in directory dir3 a version of these files with extension 
	- **eq** save the lines that are equal in both files 
	- **dif** save the lines that differ in the two files 
	- **cat** concatenates the content of the two files
```bash
#!/bin/bash

# Input checks
if [ $# -ne 4] then;
	echo "usage: $0 dir1 dir2 dir3 n"
	exit 1
fi
if [ ! -d $3 ] 
then 
	mkdir $3 
fi

# find rather than ls `find $1 -maxdepth 1 -type f -name "*.txt"`
for file in $(ls $1/*.txt); do 
	name=$(basename $file ".txt") 
	# For each .txt file in the first directory, generate the corresponding name in the second directory
	if [ -f "$2/$name.txt" ]; then 
		n1=$(cat $file | wc -l) 
		n2=$(wc –l < "$2/${name}.txt") 
		# Counts and controls the number of lines
		if [ $n1 -gt $4 -a $n2 -gt $4 ]; then
			#check for lines in file1 if they are in file2
			while read line; do 
				# Lines in file1 and file2 go in eq, lines not in file2 go in dif
				grep -q -e "^$line$" "$2/$name.txt" 
				if [ $? -eq 0 ]; then 
					echo $line >> "$3/${name}.eq" 
				else 
					echo $line >> "$3/${name}.dif" 
				fi 
			done < $file 
			
			#check for lines in file2 if they are in file1
			while read line; 
			do 
				# Inverse control for lines potentially in dif
				grep -q -e "^$line$" "$3/${name}.eq" 
				if [ $? -eq 1 ]; then 
					echo $line >> "$3/${name}.dif" 
				fi 
			done < "$2/$name.txt" 
			cat $file "$2/${name}.txt" > "$3/${name}.cat" # File concatenation
		fi 
	fi 
done	

exit 0
```
